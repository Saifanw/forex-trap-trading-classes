match /webAnalyticsEvents/{eventId} {
  // Anonymous visitors can create their own event records only.
  allow create: if request.auth != null
    && request.auth.token.firebase.sign_in_provider == 'anonymous'
    && request.resource.data.uid == request.auth.uid
    && request.resource.data.keys().hasOnly([
      'uid','pagePath','pageTitle','source','medium','campaign','referrer',
      'referrerHost','country','city','region','device','language','timezone',
      'screen','createdAt','eventAt','eventType'
    ]);
  // Only the authorized admin can read or delete analytics records.
  allow read, delete: if request.auth != null
    && request.auth.token.email == 'saif.anwar1618@gmail.com';
  // No updates to immutable event records.
  allow update: if false;
}

match /webAnalyticsSessions/{uid} {

  // Admin can read/delete sessions.
